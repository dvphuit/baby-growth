# App Shell Refactor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `app/src/App.tsx` a thin composition root by extracting sync lifecycle, modal orchestration, route composition, and modal rendering while preserving behavior and adding focused automated tests.

**Architecture:** Keep persisted UI state in the existing Zustand store and synchronization policy in `googleDriveSync.ts`. Add a React lifecycle hook for auto-sync, a hook for ephemeral modal/lightbox state, dedicated `AppRoutes` and `AppModals` wiring components, then reduce `App.tsx` to top-level composition. Add Vitest + React Testing Library + jsdom so every extracted boundary is covered independently.

**Tech Stack:** React 19, TypeScript 5.9, React Router 7, Zustand 5, Vite 7, Vitest, React Testing Library, jsdom.

## Global Constraints

- Preserve all existing user-visible behavior.
- Preserve route URLs: `/`, `/timeline`, `/growth`, `/expenses`, `/profile`, plus fallback redirect to `/`.
- Preserve Zustand persisted schemas and storage keys.
- Preserve IndexedDB behavior and keys.
- Preserve Google Drive synchronization semantics and error ownership in `googleDriveSync.ts`.
- Preserve lazy loading for currently lazy-loaded routes and modals.
- Do not redesign modal state into a state machine or discriminated-union workflow.
- Do not refactor Home, Growth, Timeline, Expenses, Profile, or unrelated feature internals.
- Do not add broad snapshot testing.
- Final validation must pass `npm test`, `npm run lint`, and `npm run build` from `app/`.

```ts
import { act, renderHook, waitFor } from '@testing-library/react';
import { beforeEach, describe, expect, it, vi } from 'vitest';
import { startAutoSync } from '@/services/googleDriveSync';
import { reloadPage } from '@/services/appRuntime';
import { useAutoSyncLifecycle } from './useAutoSyncLifecycle';

vi.mock('@/services/googleDriveSync', () => ({ startAutoSync: vi.fn() }));
vi.mock('@/services/appRuntime', () => ({ reloadPage: vi.fn() }));

const startAutoSyncMock = vi.mocked(startAutoSync);
const reloadPageMock = vi.mocked(reloadPage);

describe('useAutoSyncLifecycle', () => {
  beforeEach(() => {
    startAutoSyncMock.mockReset();
    reloadPageMock.mockReset();
  });

  it('starts auto-sync once and stops it on unmount', async () => {
    const stop = vi.fn();
    startAutoSyncMock.mockResolvedValue(stop);

    const { unmount } = renderHook(() => useAutoSyncLifecycle());
    await waitFor(() => expect(startAutoSyncMock).toHaveBeenCalledTimes(1));

    unmount();
    expect(stop).toHaveBeenCalledTimes(1);
  });

  it('stops a late auto-sync start if unmounted before startup resolves', async () => {
    const stop = vi.fn();
    let resolveStart!: (stopFn: () => void) => void;
    startAutoSyncMock.mockReturnValue(new Promise((resolve) => { resolveStart = resolve; }));

    const { unmount } = renderHook(() => useAutoSyncLifecycle());
    unmount();

    await act(async () => {
      resolveStart(stop);
    });

    expect(stop).toHaveBeenCalledTimes(1);
  });

  it('reloads when a remote update event is received', async () => {
    startAutoSyncMock.mockResolvedValue(vi.fn());
    renderHook(() => useAutoSyncLifecycle());

    window.dispatchEvent(new Event('babygrowth:remote-updated'));
    expect(reloadPageMock).toHaveBeenCalledTimes(1);
  });

  it('removes the remote update listener on unmount', async () => {
    startAutoSyncMock.mockResolvedValue(vi.fn());
    const { unmount } = renderHook(() => useAutoSyncLifecycle());
    unmount();

    window.dispatchEvent(new Event('babygrowth:remote-updated'));
    expect(reloadPageMock).not.toHaveBeenCalled();
  });

  it('does not surface a rejected startup promise from the shell hook', async () => {
    startAutoSyncMock.mockRejectedValue(new Error('sync unavailable'));
    renderHook(() => useAutoSyncLifecycle());

    await waitFor(() => expect(startAutoSyncMock).toHaveBeenCalledTimes(1));
  });
});
```

Create `app/src/hooks/useAutoSyncLifecycle.ts`:

```ts
import { useEffect } from 'react';
import { reloadPage } from '@/services/appRuntime';
import { startAutoSync } from '@/services/googleDriveSync';

export function useAutoSyncLifecycle(): void {
  useEffect(() => {
    let disposed = false;
    let stopAutoSync: (() => void) | undefined;

    void startAutoSync()
      .then((stop) => {
        if (disposed) {
          stop();
        } else {
          stopAutoSync = stop;
        }
      })
      .catch(() => {});

    const handleRemoteUpdate = () => reloadPage();
    window.addEventListener('babygrowth:remote-updated', handleRemoteUpdate);

    return () => {
      disposed = true;
      stopAutoSync?.();
      window.removeEventListener('babygrowth:remote-updated', handleRemoteUpdate);
    };
  }, []);
}
```
